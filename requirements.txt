# UniMate unified environment — one env for everything: data processing,
# rendering, training, inference and evaluation (Python 3.10, CUDA 12.4).
#
#   conda create -n unimate python=3.10 -y
#   conda activate unimate
#   pip install "setuptools<81"
#   pip install -r requirements.txt --no-build-isolation
#
# The Motion git dependency has a legacy setup.py that imports pkg_resources,
# which setuptools >= 81 no longer ships. Isolated builds always use the
# newest setuptools, so the install must run with --no-build-isolation
# against an env setuptools < 81 — installed by the step above (fresh conda
# envs ship setuptools >= 81 too); the pin below keeps it from upgrading.
#
# Only direct dependencies are listed; everything else (CUDA runtime libs,
# tokenizers, HTTP stack, ...) is resolved transitively by pip.

# PyTorch cu124 wheels (torch / torchvision "+cu124" pins below)
--extra-index-url https://download.pytorch.org/whl/cu124
# Blender official wheel index — PyPI no longer hosts cp310 bpy wheels
--extra-index-url https://download.blender.org/pypi/

setuptools<81                   # keeps pkg_resources available (Motion build)

spacy                           # T5 conditioner text normalization

openai                          # OpenAI-compatible backends (captioning; DeepSeek/GPT joint annotation)
google-genai                    # Gemini caption backend ("from google import genai")
